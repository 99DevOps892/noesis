    candidates = verify_sources(candidates)

    candidates = remove_duplicates(candidates)

    candidates = check_dependencies(candidates)

    candidates = security_filter(candidates)

    ranked = rank_capabilities(

        business_value=True,
        reliability=True,
        maintainability=True,
